cmake_minimum_required(VERSION 3.20)
project(DiveModelTests LANGUAGES CXX)
enable_testing()
add_executable(dive-model-tests DiveModelTests.cpp)
target_compile_features(dive-model-tests PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(dive-model-tests PRIVATE /W4 /EHsc)
else()
    target_compile_options(dive-model-tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME dive-model COMMAND dive-model-tests)
add_executable(coast-grid-tests CoastGridTests.cpp)
target_compile_features(coast-grid-tests PRIVATE cxx_std_17)
add_test(NAME coast-grid COMMAND coast-grid-tests)
add_executable(decompression-tests DecompressionTests.cpp)
target_compile_features(decompression-tests PRIVATE cxx_std_17)
add_test(NAME decompression COMMAND decompression-tests)
add_executable(fish-motion-tests FishMotionTests.cpp)
target_compile_features(fish-motion-tests PRIVATE cxx_std_17)
add_test(NAME fish-motion COMMAND fish-motion-tests)
